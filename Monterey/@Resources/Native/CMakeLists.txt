cmake_minimum_required(VERSION 3.24)
project(MontereyPlugins LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build the Rainmeter plugins for Windows x64.")
endif()
add_library(MontereyBluetooth SHARED MontereyBluetooth.cpp)
target_link_libraries(MontereyBluetooth PRIVATE windowsapp setupapi ole32)
add_library(MontereyBrightness SHARED Brightness/MontereyBrightness.cpp Brightness/MontereyBrightness.def)
target_link_libraries(MontereyBrightness PRIVATE wbemuuid ole32 oleaut32 dxva2 user32)
foreach(target MontereyBluetooth MontereyBrightness)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /EHsc /Brepro /Z7)
    target_link_options(${target} PRIVATE /Brepro /DEBUG:FULL "/PDBALTPATH:%_PDB%")
  endif()
endforeach()
if(BUILD_TESTING)
  add_executable(brightness_controller_test Brightness/test.cpp)
  add_executable(device_merge_test tests/device_merge_test.cpp)
  add_executable(plugin_lifecycle_test tests/plugin_lifecycle_test.cpp)
  target_link_libraries(plugin_lifecycle_test PRIVATE dbghelp)
  target_include_directories(device_merge_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  add_test(NAME brightness_controller COMMAND brightness_controller_test)
  add_test(NAME device_merge COMMAND device_merge_test)
  add_test(NAME plugin_lifecycle COMMAND plugin_lifecycle_test $<TARGET_FILE:MontereyBluetooth>)
  set_tests_properties(plugin_lifecycle PROPERTIES TIMEOUT 60)
endif()
